cmake_minimum_required(VERSION 3.16)
project(goldrop_psa_interop LANGUAGES C CXX)
enable_testing()
set(GOLDROP_PSA_SOURCE "" CACHE PATH "TF-PSA-Crypto source from ESP-IDF 6.1")
if(NOT EXISTS "${GOLDROP_PSA_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Set GOLDROP_PSA_SOURCE to ESP-IDF 6.1 components/mbedtls/mbedtls/tf-psa-crypto")
endif()
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(GEN_FILES OFF CACHE BOOL "" FORCE)
set(TF_PSA_CRYPTO_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/psa_config.h" CACHE FILEPATH "" FORCE)
add_subdirectory("${GOLDROP_PSA_SOURCE}" psa)
add_executable(goldrop_psa_interop test_crypto.cpp ../../components/management/management_crypto.cpp)
target_compile_features(goldrop_psa_interop PRIVATE cxx_std_17)
target_compile_options(goldrop_psa_interop PRIVATE -Wall -Wextra -Werror)
target_include_directories(goldrop_psa_interop PRIVATE . ../../components/model/include ../../components/management/include)
target_link_libraries(goldrop_psa_interop PRIVATE tfpsacrypto)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME goldrop_psa_crypto_interop COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_crypto.py $<TARGET_FILE:goldrop_psa_interop>)
set_tests_properties(goldrop_psa_crypto_interop PROPERTIES ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}/../../build/management-deps")
