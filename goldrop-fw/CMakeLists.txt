cmake_minimum_required(VERSION 3.22)

set(IDF_TARGET "esp32c3" CACHE STRING "Goldrop hardware target")
file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/version.txt" PROJECT_VER LIMIT_COUNT 1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/version.txt")
set(GOLDROP_RELEASE_VERSION "" CACHE STRING "Optional release version override")
if(GOLDROP_RELEASE_VERSION)
    set(PROJECT_VER "${GOLDROP_RELEASE_VERSION}")
endif()
set(GOLDROP_HARDWARE_REVISION "goldrop-c3-v1.0" CACHE STRING "Goldrop PCB revision")
if(NOT GOLDROP_HARDWARE_REVISION MATCHES "^[a-zA-Z0-9._-]+$" OR NOT PROJECT_VER MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
    message(FATAL_ERROR "Hardware revision must be an ASCII identifier; PROJECT_VER must be major.minor.patch")
endif()
string(LENGTH "${GOLDROP_HARDWARE_REVISION}" GOLDROP_HARDWARE_LENGTH)
if(GOLDROP_HARDWARE_LENGTH GREATER 31)
    message(FATAL_ERROR "Hardware revision exceeds the 31-byte image metadata limit")
endif()
string(REPLACE "." ";" GOLDROP_VERSION_PARTS "${PROJECT_VER}")
foreach(GOLDROP_VERSION_PART IN LISTS GOLDROP_VERSION_PARTS)
    string(LENGTH "${GOLDROP_VERSION_PART}" GOLDROP_VERSION_PART_LENGTH)
    if(GOLDROP_VERSION_PART_LENGTH GREATER 9 OR GOLDROP_VERSION_PART MATCHES "^0[0-9]")
        message(FATAL_ERROR "Version components must have at most 9 digits and no leading zeroes")
    endif()
endforeach()
add_compile_definitions(GOLDROP_FIRMWARE_VERSION="${PROJECT_VER}" GOLDROP_HARDWARE_REVISION="${GOLDROP_HARDWARE_REVISION}")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
project(goldrop)

# Existing sdkconfig files override sdkconfig.defaults. Reject an older small
# startup stack before shipping an image that performs the larger NVS migration.
if(CONFIG_ESP_MAIN_TASK_STACK_SIZE LESS 8192)
    message(FATAL_ERROR "Goldrop requires CONFIG_ESP_MAIN_TASK_STACK_SIZE>=8192; update your local sdkconfig (defaults do not override existing values)")
endif()

if(NOT IDF_TARGET STREQUAL "esp32c3")
    message(FATAL_ERROR "Goldrop requires ESP32-C3")
endif()
if(DEFINED GOLDROP_PIO_CONFIG_FINGERPRINT)
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/goldrop-configure.sha256"
         CONTENT "${GOLDROP_PIO_CONFIG_FINGERPRINT}\n")
endif()
